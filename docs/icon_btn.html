Icon Button

Square buttons for icon-only actions.

Base

Mute Variant

Sizes

States

API Reference

Classes

Class Description
icon-btn Base icon button
icon-btn:mute Muted variant

Modifiers

Modifier Description
-sm Small size

States

Attribute Description
disabled Disabled state

Structure

[class~="icon-btn"], [class*="icon-btn:"] {
  /* Base styles */
  
  /* Variant */
  &[class*=":mute"] { }
  
  /* Size */
  &[class*="-sm"] { }
  
  /* States */
  &[aria-busy="true"] { }
  &:disabled { }
}

Usage

<!-- Base -->
<button class="icon-btn" aria-label="Menu">
  <svg>...</svg>
</button>

<!-- Variant -->
<button class="icon-btn:mute" aria-label="Settings">
  <svg>...</svg>
</button>

<!-- Variant + Size -->
<button class="icon-btn:mute-sm" aria-label="Close">
  <svg>...</svg>
</button>
