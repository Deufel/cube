Usage

<!-- Composition + Utilities -->
<div class="flex:stack g-md p-lg">
  <header>...</header>
  <main>...</main>
</div>

<!-- Centered card content -->
<div class="card flex:stack g-sm p-md align:center justify:center">
  <span>Icon</span>
  <span>Label</span>
</div>

<!-- Responsive grid with gap -->
<div class="grid:auto-fit g-lg" style="--grid-min: 200px;">
  <div class="card p-md">Card 1</div>
  <div class="card p-md">Card 2</div>
</div>

<!-- Header with items pushed apart -->
<header class="flex:cluster justify:between align:center g-md p-sm">
  <h1>Title</h1>
  <nav>Links</nav>
</header>
