API Reference

Classes

Class Description
btn Base (outlined secondary)
btn:pri Primary
btn:sec Secondary
btn:inf Info
btn:suc Success
btn:wrn Warning
btn:dgr Danger

Modifiers

Modifier Description
-sm Small size
-lg Large size
out Outline style

States

Attribute Description
disabled Disabled
aria-busy Loading

Structure

[class~="btn"], [class*="btn:"] {
  /* Base styles */
  
  /* States */
  &[aria-busy="true"] { }
  &:disabled { }
  
  /* Sizes */
  &[class*="-sm"] { }
  &[class*="-lg"] { }
  
  /* Variants */
  &[class*=":pri"] { }
  &[class*=":sec"] { }
  &[class*=":wrn"] { }
  &[class*=":dgr"] { }
  
  /* Outline modifier */
  &[class~="out"] { }
}

Usage

<!-- Base -->
<button class="btn">Default</button>

<!-- Variant -->
<button class="btn:pri">Primary</button>

<!-- Variant + Size -->
<button class="btn:pri-sm">Small Primary</button>

<!-- Variant + Outline -->
<button class="btn:dgr out">Danger Outline</button>

<!-- States -->
<button class="btn:pri" disabled>Disabled</button>
<button class="btn:pri" aria-busy="true">Loading</button>
