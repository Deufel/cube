Dialogs & Drawers

Modal dialogs and slide-out drawer panels.

Centered Dialog

Drawers

API Reference

Dialog

Element Description
<dialog> Native dialog
::backdrop 50% dark overlay

Drawers

Class Description
drawer:left From left
drawer:right From right
drawer:top From top
drawer:bottom From bottom

Usage

<!-- Centered Dialog -->
<dialog id="my-dialog">
  <div class="card flex:stack gap:sm p:md">
    <h2>Dialog Title</h2>
    <p>Content here.</p>
    <button data-on:click="this.closest('dialog').close()">Close</button>
  </div>
</dialog>

<!-- Drawer -->
<dialog id="my-drawer" class="drawer:left">
  <div class="card gla flex:stack gap:sm p:md">
    <h2>Drawer Title</h2>
    <button data-on:click="this.closest('dialog').close()">Close</button>
  </div>
</dialog>

<!-- Open with Datastar -->
<button data-on:click="document.getElementById('my-dialog').showModal()">
  Open Dialog
</button>
