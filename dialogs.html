Dialogs & Drawers

Modal dialogs and slide-out drawer panels.

API Reference

Dialog

Element Description
<dialog> Native dialog
::backdrop 50% dark overlay

Drawers

Class Description
drawer:left From left
drawer:right From right
drawer:top From top
drawer:bottom From bottom

Usage

<!-- Centered Dialog with click-outside-to-close -->
<!-- Use kebab-case in data-ref (auto-converts to camelCase signal) -->
<dialog data-ref:my-dialog
        data-on:click="evt.target === el && el.close()">
  <div class="card flex:stack gap:sm p:md">
    <h2>Dialog Title</h2>
    <p>Content here.</p>
    <button data-on:click="el.closest('dialog').close()">Close</button>
  </div>
</dialog>

<!-- Drawer -->
<dialog data-ref:my-drawer class="drawer:left"
        data-on:click="evt.target === el && el.close()">
  <div class="flex:stack gap:sm p:md">
    <h2>Drawer Title</h2>
    <button data-on:click="el.closest('dialog').close()">Close</button>
  </div>
</dialog>

<!-- Open with Datastar (signals are camelCase) -->
<button data-on:click="$myDialog.showModal()">Open Dialog</button>
<button data-on:click="$myDrawer.showModal()">Open Drawer</button>
